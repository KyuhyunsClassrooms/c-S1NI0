# 1. 프로그램 설계서 (5점)

## 1. 프로젝트 주제
규칙 기반 에이전트 - 틱택토

## 2. 핵심 기능 목록
1.  3 x 3 판에서 플레이어와 프로그램이 O와 X를 번갈아 둔다
2.  플레이어가 두는 수에 따라 규칙에 맞춰 배열한다.
3.  규칙은 다음과 같다.
🧩 규칙 1 — 이길 수 있으면 이긴다
보드에 자신(X)의 2개와 빈칸 하나가 있으면 그 칸에 둔다.
🧩 규칙 2 — 상대의 승리를 막는다
보드에 상대(O)의 2개와 빈칸 하나가 있으면 그 칸에 둔다.
🧩 규칙 3 — 선공일 때는 반드시 중앙을 잡는다
중앙이 비어 있으면 무조건 중앙에 둔다.
🧩 규칙 4 — 상대가 중앙을 뺏었으면 코너를 잡는다
🧩 규칙 5 — 포크(Fork)를 만든다 (공격용)
자신이 두 방향으로 동시에 승리할 수 있는 포크를 만들 수 있으면 만든다.
🧩 규칙 6 — 상대의 포크를 막는다 (방어용)
상대가 포크를 만들 수 있는 위치가 있으면, 그 포크를 막거나 상대가 포크를 만들 수 없도록 다른 곳에 둔다.
🧩 규칙 7 — 중앙이 이미 차지되었으면 반대편 코너를 잡는다
🧩 규칙 8 — 빈 코너가 있으면 코너에 둔다
🧩 규칙 9 — 그 외에는 빈 변(edge)에 둔다

#include <stdio.h>

#define SIZE 3

char board[SIZE][SIZE];
char player, agent;

// ===== 보드 관련 기본 함수 =====

void initBoard() {
    for (int i = 0; i < SIZE; i++)
        for (int j = 0; j < SIZE; j++)
            board[i][j] = ' ';
}

void printBoard() {
    printf("\n");
    for (int i = 0; i < SIZE; i++) {
        for (int j = 0; j < SIZE; j++) {
            printf(" %c ", board[i][j]);
            if (j < SIZE - 1) printf("|");
        }
        printf("\n");
        if (i < SIZE - 1) printf("---+---+---\n");
    }
    printf("\n");
}

int checkWin(char p) {
    for (int i = 0; i < SIZE; i++) {
        if (board[i][0] == p && board[i][1] == p && board[i][2] == p) return 1;
        if (board[0][i] == p && board[1][i] == p && board[2][i] == p) return 1;
    }
    if (board[0][0] == p && board[1][1] == p && board[2][2] == p) return 1;
    if (board[0][2] == p && board[1][1] == p && board[2][0] == p) return 1;
    return 0;
}

int isFull() {
    for (int i = 0; i < SIZE; i++)
        for (int j = 0; j < SIZE; j++)
            if (board[i][j] == ' ') return 0;
    return 1;
}

// ===== 규칙 기반 에이전트 함수 =====

// 규칙 1, 2: 승리 또는 방어
int tryWinOrBlock(char target) {
    for (int i = 0; i < SIZE; i++) {
        for (int j = 0; j < SIZE; j++) {
            if (board[i][j] == ' ') {
                board[i][j] = target;
                if (checkWin(target)) {
                    board[i][j] = agent;
                    return 1;
                }
                board[i][j] = ' ';
            }
        }
    }
    return 0;
}

// 규칙 8: 코너 선택
int takeCorner() {
    int corners[4][2] = {{0,0},{0,2},{2,0},{2,2}};
    for (int i = 0; i < 4; i++) {
        int r = corners[i][0], c = corners[i][1];
        if (board[r][c] == ' ') {
            board[r][c] = agent;
            return 1;
        }
    }
    return 0;
}

// 규칙 9: 변(edge) 선택
int takeEdge() {
    int edges[4][2] = {{0,1},{1,0},{1,2},{2,1}};
    for (int i = 0; i < 4; i++) {
        int r = edges[i][0], c = edges[i][1];
        if (board[r][c] == ' ') {
            board[r][c] = agent;
            return 1;
        }
    }
    return 0;
}

void agentMove() {
    // 규칙 1: 이길 수 있으면 이기기
    if (tryWinOrBlock(agent)) return;

    // 규칙 2: 막을 수 있으면 막기
    if (tryWinOrBlock(player)) return;

    // 규칙 3: 중앙 잡기
    if (board[1][1] == ' ') {
        board[1][1] = agent;
        return;
    }

    // 규칙 4: 코너 잡기
    if (takeCorner()) return;

    // 규칙 5: 변 잡기
    if (takeEdge()) return;
}
